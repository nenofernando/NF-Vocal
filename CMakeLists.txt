cmake_minimum_required(VERSION 3.22)
project(NFVocal VERSION 1.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET 10.13 CACHE STRING "")
set(JUCE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/JUCE" CACHE PATH "JUCE source")
add_subdirectory("${JUCE_SOURCE_DIR}" JUCE)
juce_add_binary_data(NFAssets SOURCES Assets/Logo.png Assets/Signature.png)
juce_add_plugin(NFVocal COMPANY_NAME "NF Audio Tools" BUNDLE_ID "com.nfaudiotools.vocal" PLUGIN_MANUFACTURER_CODE Nfat PLUGIN_CODE Nfvc FORMATS VST3 AU PRODUCT_NAME "NF Vocal" AU_MAIN_TYPE kAudioUnitType_Effect VST3_CATEGORIES Fx Vocals IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(NFVocal)
target_sources(NFVocal PRIVATE Source/Plugin.cpp Source/Plugin.h)
target_compile_definitions(NFVocal PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(NFVocal PRIVATE NFAssets juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
add_executable(NFVocalCheck Source/Check.cpp)
target_include_directories(NFVocalCheck PRIVATE $<TARGET_PROPERTY:NFVocal,INCLUDE_DIRECTORIES>)
target_compile_definitions(NFVocalCheck PRIVATE $<TARGET_PROPERTY:NFVocal,COMPILE_DEFINITIONS>)
target_link_libraries(NFVocalCheck PRIVATE NFVocal NFAssets)
